20 // Pixel and tile rectangles are half-open. Grids share coordinate origin zero;

62 template <typename T>
63 Rectangle<T> computeTileLocalIntersection(Rectangle<T> layerArea, Rectangle<int> tileBounds)
64 {
65 const Rectangle<T> bounds(tileBounds);
66 if(!bounds.intersects(layerArea))
67 {
68 return {};
69 }
70 return bounds.intersection(layerArea) - bounds.getTopLeft();
71 }
