371 auto schedule(auto&&... params)
372 {
373 using Builder = Scroom::Detail::ThreadPool::Builder<false, false, false, void>;
374
375 auto r = Builder(defaultPriority).addMany(std::forward<decltype(params)>(params)...);
376
377 static_assert(r.hasCallable);
378
379 if constexpr(!r.hasQueue)
380 {
381 r.queue = defaultQueue()->getWeak();
382 }
383
384 using R = decltype(r)::ResultType;
385
386 if constexpr(std::is_void_v<R>)
387 {
388 schedule_impl(std::move(r.fn), r.priority, std::move(r.queue));
389 }
390 else
391 {
392 boost::packaged_task<R> t(std::move(r.fn));
393 boost::unique_future<R> f = t.get_future();
394 schedule_impl(std::move(t), r.priority, std::move(r.queue));
395 return f;
396 }
