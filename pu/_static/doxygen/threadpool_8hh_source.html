380 auto schedule(auto&&... params)
381 {
382 using Builder = Scroom::Detail::ThreadPool::Builder<false, false, false, void>;
383
384 auto r = Builder(defaultPriority).addMany(std::forward<decltype(params)>(params)...);
385
386 static_assert(r.hasCallable);
387
388 if constexpr(!r.hasQueue)
389 {
390 r.queue = defaultQueue()->getWeak();
391 }
392
393 using R = decltype(r)::ResultType;
394
395 if constexpr(std::is_void_v<R>)
396 {
397 schedule_impl(std::move(r.fn), r.priority, std::move(r.queue));
398 }
399 else
400 {
401 boost::packaged_task<R> t(std::move(r.fn));
402 boost::unique_future<R> f = t.get_future();
403 schedule_impl(std::move(t), r.priority, std::move(r.queue));
