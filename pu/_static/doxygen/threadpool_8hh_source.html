401 auto schedule(auto&&... params)
402 {
403 using Builder = Scroom::Detail::ThreadPool::Builder<false, false, false, void>;
404
405 auto r = Builder(defaultPriority).addMany(std::forward<decltype(params)>(params)...);
406
407 static_assert(r.hasCallable);
408
409 if constexpr(!r.hasQueue)
410 {
411 r.queue = defaultQueue()->getWeak();
412 }
413
414 using R = decltype(r)::ResultType;
415
416 if constexpr(std::is_void_v<R>)
417 {
418 schedule_impl(std::move(r.fn), r.priority, std::move(r.queue));
419 }
420 else
421 {
422 boost::packaged_task<R> t(std::move(r.fn));
423 boost::unique_future<R> f = t.get_future();
424 schedule_impl(std::move(t), r.priority, std::move(r.queue));
425 return f;
426 }
