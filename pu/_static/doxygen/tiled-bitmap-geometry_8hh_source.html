53 Rectangle<T> intersectionInTileCoordinates(Rectangle<T> areaInLayerCoordinates, Rectangle<int> tileAreaInLayerCoordinates)
54 {
55 const Rectangle<T> tileBoundsInLayerCoordinates(tileAreaInLayerCoordinates);
56 if(!tileBoundsInLayerCoordinates.intersects(areaInLayerCoordinates))
57 {
58 return {};
59 }
60 return tileBoundsInLayerCoordinates.intersection(areaInLayerCoordinates) - tileBoundsInLayerCoordinates.getTopLeft();
61 }
