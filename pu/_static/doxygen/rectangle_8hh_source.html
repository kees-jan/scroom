21 template <IntegralCoordinate T>
22 class RectangleIterator : public boost::addable2<RectangleIterator<T>, std::ptrdiff_t>
23 {
24 public:
25 using value_type = Point<T>;
26 using difference_type = std::ptrdiff_t;
27 using iterator_concept = std::forward_iterator_tag;
28 using iterator_category = std::input_iterator_tag;
29
30 RectangleIterator() = default;
31 RectangleIterator(T left, T top, T right, T bottom)
32 : m_left(left)
33 , m_right(right)
34 , m_bottom(bottom)
35 , m_x(left)
36 , m_y(left == right ? bottom : top)
37 {
38 }
39
40 Point<T> operator*() const { return {m_x, m_y}; }
41 RectangleIterator& operator++()
42 {
43 if(++m_x == m_right)
44 {
45 m_x = m_left;
46 ++m_y;
47 }
48 return *this;
49 }
50 RectangleIterator operator++(int)
51 {
52 auto previous = *this;
53 ++*this;
54 return previous;
55 }
56
58 RectangleIterator& operator+=(difference_type offset)
59 {
60 const auto width = static_cast<difference_type>(m_right - m_left);
61 auto rows = offset / width;
62 const auto column = static_cast<difference_type>(m_x - m_left) + (offset % width);
63 rows += column / width;
64 m_x = static_cast<T>(m_left + column % width);
65 m_y = static_cast<T>(m_y + rows);
66 return *this;
67 }
68
69 bool operator==(const RectangleIterator& other) const
70 {
71 return m_left == other.m_left && m_right == other.m_right && m_bottom == other.m_bottom && m_x == other.m_x
72 && m_y == other.m_y;
73 }
74 bool operator==(std::default_sentinel_t /*sentinel*/) const { return m_y >= m_bottom; }
75
76 private:
77 T m_left{};
78 T m_right{};
79 T m_bottom{};
80 T m_x{};
81 T m_y{};
82 };
