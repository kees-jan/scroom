46template <typename T>
47class Freezable
48{
49public:
50 using value_type = T;
51 using Me = Freezable<T>;
52
53private:
54 value_type value;
55 unsigned locked{0};
56
57public:
58 Freezable() = default;
59
60 explicit Freezable(value_type v)
61 : value(std::move(v))
62 {
63 }
64
65 [[nodiscard]] const value_type& get() const { return value; }
66
67 bool set(value_type v)
68 {
69 if(!locked)
70 {
71 value = std::move(v);
72 }
73 return locked;
74 }
75
76 void lock() { locked++; }
77
78 void unlock()
79 {
80 require(locked > 0);
81 locked--;
82 }
83 [[nodiscard]] bool is_locked() const { return locked; }
84
85 operator const value_type&() const { return value; } // NOLINT(hicpp-explicit-conversions)
86 Me& operator=(value_type v)
87 {
88 set(std::move(v));
89 return *this;
90 }
91 const value_type* operator*() const { return &value; }
92 const value_type* operator->() const { return &value; }
93 Me& operator+=(value_type v) { return operator=(value + v); }
94 Me& operator-=(value_type v) { return operator=(value - v); }
95};
